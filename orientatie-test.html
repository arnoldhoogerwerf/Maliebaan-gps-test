function renderLoop() { // 1. Smooth Kompas Heading (met correctie voor de 0/360-grens) let diff = rawHeading - smoothedHeading; // Dwing het verschil altijd tussen -180 en +180 graden diff = Math.atan2(Math.sin(diff * Math.PI / 180), Math.cos(diff * Math.PI / 180)) * 180 / Math.PI; smoothedHeading += diff * 0.1; // Smoothing factor // Zorg dat smoothedHeading netjes tussen 0 en 360 blijft smoothedHeading = (smoothedHeading + 360) % 360; if (userCoords && targetLocations.length > 0) { targetLocations.forEach((target, i) => { const arrowEl = document.getElementById(`arrow-${i}`); const labelEl = document.getElementById(`label-${i}`); // Smooth Afstand arrowData[i].currentDist += (arrowData[i].targetDist - arrowData[i].currentDist) * 0.05; const dist = arrowData[i].currentDist; // Schaal Pijllengte const arrowLength = Math.max(20, Math.min(130, (dist / 150) * 130)); // 2. Smooth Rotatie van de pijl (ook met 0/360 correctie) const targetAngle = arrowData[i].bearing - smoothedHeading; let angleDiff = targetAngle - arrowData[i].currentAngle; // Dwing de hoek van de pijl ook over de kortste weg angleDiff = Math.atan2(Math.sin(angleDiff * Math.PI / 180), Math.cos(angleDiff * Math.PI / 180)) * 180 / Math.PI; arrowData[i].currentAngle += angleDiff * 0.15; // Pas CSS toe arrowEl.style.height = `${arrowLength}px`; arrowEl.style.transform = `rotate(${arrowData[i].currentAngle}deg)`; labelEl.innerText = `${Math.round(dist)}m`; }); } requestAnimationFrame(renderLoop); }
